1. NODE LAIN - ROUTER

Nama script: setup_lain.sh

#!/bin/sh

echo "=== Setup Router Lain ==="

ip link set eth0 up
ip link set eth1 up
ip link set eth2 up
ip link set eth3 up

ip addr replace 10.75.1.1/24 dev eth1
ip addr replace 10.75.2.1/24 dev eth2
ip addr replace 10.75.3.1/24 dev eth3

udhcpc -i eth0 2>/dev/null || true

echo 1 > /proc/sys/net/ipv4/ip_forward

iptables -t nat -C POSTROUTING -o eth0 -j MASQUERADE 2>/dev/null || \
iptables -t nat -A POSTROUTING -o eth0 -j MASQUERADE

iptables -C FORWARD -i eth1 -o eth0 -j ACCEPT 2>/dev/null || \
iptables -A FORWARD -i eth1 -o eth0 -j ACCEPT

iptables -C FORWARD -i eth2 -o eth0 -j ACCEPT 2>/dev/null || \
iptables -A FORWARD -i eth2 -o eth0 -j ACCEPT

iptables -C FORWARD -i eth3 -o eth0 -j ACCEPT 2>/dev/null || \
iptables -A FORWARD -i eth3 -o eth0 -j ACCEPT

iptables -C FORWARD -i eth0 -o eth1 -m state --state RELATED,ESTABLISHED -j ACCEPT 2>/dev/null || \
iptables -A FORWARD -i eth0 -o eth1 -m state --state RELATED,ESTABLISHED -j ACCEPT

iptables -C FORWARD -i eth0 -o eth2 -m state --state RELATED,ESTABLISHED -j ACCEPT 2>/dev/null || \
iptables -A FORWARD -i eth0 -o eth2 -m state --state RELATED,ESTABLISHED -j ACCEPT

iptables -C FORWARD -i eth0 -o eth3 -m state --state RELATED,ESTABLISHED -j ACCEPT 2>/dev/null || \
iptables -A FORWARD -i eth0 -o eth3 -m state --state RELATED,ESTABLISHED -j ACCEPT

echo "=== Konfigurasi Lain selesai ==="
ip -br addr
ip route
iptables -t nat -L -n -v

        -N "" \

Nama script: setup_chisa.sh

#!/bin/sh

echo "=== Setup Chisa ==="

ip link set eth0 up
ip addr replace 10.75.2.2/24 dev eth0

ip route del default 2>/dev/null || true
ip route add default via 10.75.2.1

echo "nameserver 8.8.8.8" > /etc/resolv.conf

if ! command -v vsftpd >/dev/null 2>&1; then
    apk update
    apk add vsftpd
fi

id alice >/dev/null 2>&1 || adduser -D alice
id mika  >/dev/null 2>&1 || adduser -D mika
id eiri  >/dev/null 2>&1 || adduser -D eiri

# Password diatur manual:
# passwd alice
# passwd mika
# passwd eiri

getent group wiredread >/dev/null 2>&1 || addgroup wiredread
addgroup mika wiredread 2>/dev/null || true

mkdir -p /var/wired/data
chown alice:wiredread /var/wired/data
chmod 2750 /var/wired/data

mkdir -p /etc/vsftpd
mkdir -p /etc/vsftpd/user_conf

cat > /etc/vsftpd/vsftpd.conf <<'EOF'
listen=YES
listen_ipv6=NO
anonymous_enable=NO
local_enable=YES
write_enable=YES
local_umask=022
chroot_local_user=YES
allow_writeable_chroot=YES
local_root=/var/wired/data
userlist_enable=YES
userlist_deny=YES
userlist_file=/etc/vsftpd/user_list
pasv_enable=YES
seccomp_sandbox=NO
cmds_denied=EPSV
user_config_dir=/etc/vsftpd/user_conf
EOF

echo "eiri" > /etc/vsftpd/user_list
echo "write_enable=NO" > /etc/vsftpd/user_conf/mika

pkill vsftpd 2>/dev/null || true
/usr/sbin/vsftpd /etc/vsftpd/vsftpd.conf &

echo ""
echo "=== VSFTPD Configuration ==="
cat /etc/vsftpd/vsftpd.conf

echo ""
echo "FTP data directory:"
ls -ld /var/wired/data

echo ""
echo "=== Chisa selesai ==="

Nama script: setup_knights.sh

#!/bin/sh

echo "=== Setup Knights ==="

ip link set eth0 up
ip addr replace 10.75.3.2/24 dev eth0

ip route del default 2>/dev/null || true
ip route add default via 10.75.3.1

echo "nameserver 8.8.8.8" > /etc/resolv.conf

if ! command -v sshd >/dev/null 2>&1; then
    apk update
    apk add openssh
fi

ssh-keygen -A
mkdir -p /run/sshd

if ! id mika_admin >/dev/null 2>&1; then
    adduser -D mika_admin
fi

mkdir -p /home/mika_admin/.ssh
chmod 700 /home/mika_admin/.ssh

cat > /home/mika_admin/.ssh/authorized_keys <<'EOF'
ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIHiOtTUY8V6P5WLZjvzXLUpsHyho9jClyBJ/vN+xyMCV mika_admin@Mika
EOF

chmod 600 /home/mika_admin/.ssh/authorized_keys
chown -R mika_admin:mika_admin /home/mika_admin/.ssh

if grep -q '^[#[:space:]]*PubkeyAuthentication' /etc/ssh/sshd_config; then
    sed -i 's/^[#[:space:]]*PubkeyAuthentication.*/PubkeyAuthentication yes/' /etc/ssh/sshd_config
else
    echo 'PubkeyAuthentication yes' >> /etc/ssh/sshd_config
fi

if grep -q '^[#[:space:]]*PasswordAuthentication' /etc/ssh/sshd_config; then
    sed -i 's/^[#[:space:]]*PasswordAuthentication.*/PasswordAuthentication no/' /etc/ssh/sshd_config
else
    echo 'PasswordAuthentication no' >> /etc/ssh/sshd_config
fi

/usr/sbin/sshd -t || {
    echo "ERROR: sshd_config tidak valid."
    exit 1
}

pkill sshd 2>/dev/null || true
/usr/sbin/sshd

echo ""
echo "SSH effective configuration:"
/usr/sbin/sshd -T | grep -E 'pubkeyauthentication|passwordauthentication'

echo ""
echo "Port 22:"
ss -lntp | grep :22

echo ""
echo "=== Knights selesai ==="
